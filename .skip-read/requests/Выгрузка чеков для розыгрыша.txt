Выгрузка чеков для розыгрыша

-- Список принятых чеков с данными участников за период {{startDate}} — {{endDate}}.
-- Период фильтруется по дате регистрации (created_at); в выдаче — дата загрузки (created_at).
SELECT
    BIN_TO_UUID(r.id)     AS receipt_id,
    r.created_at          AS uploaded_at,
    u.participant_code,
    u.email,
    u.phone,
    BIN_TO_UUID(u.id)     AS user_id,
    ROUND(r.sum / 100, 2) AS sum_rub
FROM receipt r
JOIN `user` u ON u.id = r.user_id
WHERE r.status = 'ACCEPTED'
  AND r.created_at >= {{startDate}}
  AND r.created_at <= {{endDate}}
ORDER BY r.sum ASC;
